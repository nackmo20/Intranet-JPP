  // Lit l'état sauvegardé ; si le JSON est cassé, on repart d'un état propre.
  function loadState() {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) {
      return initialState();
    }
    try {
      return { ...initialState(), ...JSON.parse(raw) };
    } catch (error) {
      console.warn('[EAFC automation] État illisible, réinitialisation.', error);
      return initialState();
    }
  }

      autoResume: false,
      delayMs: 900,

  }

  // Bouton "Charger" : parse l'export create_pages collé dans le panneau.
  function loadExportFromPanel() {
    try {
      const text = document.querySelector('#eafc-auto-panel [data-role="json"]').value.trim();
      if (!text) {
        alert('Collez un export JSON avant de charger.');
        return;
      }
      const data = parseExportJson(text);
      if (data.type !== 'create_pages' || !Array.isArray(data.items)) {
        throw new Error('Export invalide : type=create_pages et items[] attendus.');
      }
      state.exportData = data;
      state.index = 0;
      state.autoResume = false;
      state.createdRows = [];
      state.pendingCreatedRow = null;
      saveState();
      log(`${data.items.length} parcours chargés depuis l'export.`);
    } catch (error) {
      console.error('[EAFC automation] Chargement impossible', error);
      log(`Chargement impossible : ${error.message}`);
      alert(`Chargement impossible : ${error.message}`);
    }
  }

  function parseExportJson(text) {
    const withoutFence = text
      .replace(/^```(?:json)?/i, '')
      .replace(/```$/i, '')
      .trim();
    try {
      return JSON.parse(withoutFence);
    } catch (error) {
      const start = withoutFence.indexOf('{');
      const end = withoutFence.lastIndexOf('}');
      if (start === -1 || end === -1 || end <= start) throw error;
      return JSON.parse(withoutFence.slice(start, end + 1));
    }
  }


  // Exporte l'historique des parcours créés dans un fichier .xls ouvrable avec Excel/LibreOffice.
  function exportCreatedRowsToExcel() {
    recordCreatedPageIfAvailable();
    const rows = [...(state.createdRows || [])];
    if (state.pendingCreatedRow) rows.push({ ...state.pendingCreatedRow, status: 'En attente de récupération ID/lien' });
    if (!rows.length) {
      alert('Aucune création enregistrée à exporter pour le moment.');
      return;
    }
    const headers = [
      'Statut', 'Node ID Drupal', 'Lien de la page', 'Titre', 'Chapo', 'Fiche formation export OVP', 'Thématiques saisies', 'Métiers',
      'Départements', 'Début préinscription', 'Dépublication', 'Identifiant parcours export', 'Source OVP', 'Plan', 'Date export',
    ];
    const html = `<!doctype html><html><head><meta charset="utf-8"><style>td{white-space:pre-wrap;}</style></head><body><table><thead><tr>${headers.map(header => `<th>${escapeHtml(header)}</th>`).join('')}</tr></thead><tbody>${rows.map(row => `<tr>${[
      row.status || 'Créé', row.nodeId || '', row.pageUrl || '', row.title || '', row.chapo || '', row.ficheFormation || '', row.thematics || '', row.publics || '',
      row.departmentCodes || '', row.preinscriptionStartDate || '', row.unpublishDate || '', row.drupalParcoursId || '', row.sourceOvpId || '', row.planTitle || '', row.exportedAt || '',
    ].map(value => `<td>${escapeHtml(value)}</td>`).join('')}</tr>`).join('')}</tbody></table></body></html>`;
    const blob = new Blob([html], { type: 'application/vnd.ms-excel;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `parcours-drupal-${new Date().toISOString().slice(0, 10)}.xls`;
    document.body.appendChild(link);
    link.click();
    link.remove();
    URL.revokeObjectURL(url);
    log(`${rows.length} ligne(s) exportée(s) vers Excel.`);
  }

  // Prépare une ligne d'export avec toutes les valeurs saisies avant le clic Enregistrer.
  function buildCreatedRow(item) {
    return {
      status: 'Soumis',
      nodeId: '',
      pageUrl: '',
      title: item.title || '',
      chapo: item.chapo || item.content || '',
      ficheFormation: ficheFormationText(item),
      thematics: resolveThematics(item).join(', '),
      publics: uniquePublics([...(state.exportData.publics || []), ...(item.publics || [])]).map(value => publicPath(value).join(' > ')).join(' | '),
      departmentCodes: (item.departmentCodes?.length ? item.departmentCodes : String(item.departmentCode || '').split('/')).filter(Boolean).join(', '),
      preinscriptionStartDate: item.defaultDates?.preinscriptionStartDate || state.exportData.defaultDates?.preinscriptionStartDate || '',
      unpublishDate: item.defaultDates?.unpublishDate || state.exportData.defaultDates?.unpublishDate || '',
      drupalParcoursId: item.drupalParcoursId || '',
      sourceOvpId: item.sourceOvpId || '',
      planTitle: state.exportData.planTitle || '',
      exportedAt: new Date().toISOString(),
    };
  }

  // Après l'enregistrement, récupère l'ID node et le lien public depuis la barre Seeds/Drupal, puis complète la ligne Excel.
  function recordCreatedPageIfAvailable() {
    if (!state.pendingCreatedRow) return;
    const nodeInfo = getCurrentNodeInfo();
    if (!nodeInfo.nodeId && !nodeInfo.pageUrl) return;
    state.createdRows = [...(state.createdRows || []), {
      ...state.pendingCreatedRow,
      status: 'Créé',
      nodeId: nodeInfo.nodeId,
      pageUrl: nodeInfo.pageUrl,
    }];
    state.pendingCreatedRow = null;
    saveState();
    log(`Parcours Drupal enregistré dans l’export Excel : node ${nodeInfo.nodeId || 'inconnu'}.`);
  }

  // Recherche l'identifiant (ex: node/1349) et le lien "Voir" de la page créée.
  function getCurrentNodeInfo() {
    const nodeLinks = [...document.querySelectorAll('a[href], a[data-drupal-link-system-path]')];
    const paths = nodeLinks.flatMap(link => [link.getAttribute('data-drupal-link-system-path'), link.getAttribute('href')]).filter(Boolean);
    const nodeId = (paths.map(path => path.match(/(?:^|\/)node\/(\d+)(?:\/|$|[?#])/)?.[1]).find(Boolean)) || '';
    const viewLink = nodeLinks.find(link => link.getAttribute('data-drupal-link-system-path') === `node/${nodeId}`)
      || nodeLinks.find(link => normalize(link.textContent) === 'voir' && link.getAttribute('href'));
    const pageUrl = viewLink?.href || (nodeId && /\/node\/\d+/.test(window.location.pathname) ? window.location.href : '');
    return { nodeId, pageUrl };
  }

  // Point d'entrée principal : ouvre le formulaire ou remplit le formulaire Formation courant.
  async function startAutomation() {
    if (running) return;
    if (!state.exportData?.items?.length) {
      alert('Chargez un export Drupal avant de démarrer.');
      return;
    }
    if (state.index >= state.exportData.items.length) {
      log('Tous les parcours ont déjà été traités.');
      state.autoResume = false;
      saveState();
      return;
    }
    running = true;
    paused = false;
    state.autoResume = true;
    saveState();

    try {
      if (isFormationForm()) {
        await fillCurrentFormation();
      } else {
        await goToFormationForm();
      }
    } catch (error) {
      console.error('[EAFC automation]', error);
      log(`Erreur : ${error.message}`);
      running = false;
      state.autoResume = false;
      saveState();
      alert(`Automation interrompue : ${error.message}`);
    }
  }

  // Navigation initiale : clique Seeds Add puis Formation, ou directement Formation si le lien est déjà visible.
  async function goToFormationForm() {
    const item = currentItem();
    log(`Ouverture du formulaire Formation pour : ${item.title}`);
    const directLink = document.querySelector(SELECTORS.formationLink);
    if (directLink) {
      await safeClick(directLink, 'Cliquer sur Formation');
      return;
    }
    const seedsButton = await waitForElement(SELECTORS.seedsAdd, 6000);
    await safeClick(seedsButton, 'Cliquer sur Seeds Add');
    const formationLink = await waitForElement(SELECTORS.formationLink, 10000);
    await safeClick(formationLink, 'Cliquer sur Formation');
  }

  // Orchestration d'un parcours : tous les champs Drupal sont remplis dans l'ordre demandé, puis Enregistrer.
  async function fillCurrentFormation() {
    const item = currentItem();
    log(`Remplissage du parcours ${state.index + 1}/${state.exportData.items.length} : ${item.title}`);

    // 1) Titre et chapo.
    setValue(await waitForElement(SELECTORS.title), truncate(item.title || '', 100));
    setValue(await waitForElement(SELECTORS.chapo), item.chapo || item.content || '');

    // 2) Image, thématiques, publics, géographie, dates et paragraphes.
    await selectRandomMedia();
    await fillThematics(item);
    await fillPublics(item);
    await fillGeography(item);
    setValue(await waitForElement(SELECTORS.preinscriptionDate), toHtmlDate(item.defaultDates?.preinscriptionStartDate || state.exportData.defaultDates?.preinscriptionStartDate));
    await addRichtextParagraph(item);
    await addReturnTrainingLink();
    setValue(await waitForElement(SELECTORS.unpublishDate), toHtmlDate(item.defaultDates?.unpublishDate || state.exportData.defaultDates?.unpublishDate));
    const unpublishState = await waitForElement(SELECTORS.unpublishState);
    await safeClick(unpublishState, 'Cliquer sur l’état de dépublication');
    setSelectValue(unpublishState, 'unpublished', 'Dépublié');

    // 3) Enregistrement : l'index est incrémenté juste avant le clic pour reprendre au bon item.
    const submit = await waitForElement(SELECTORS.submit);
    await safeClick(submit, 'Enregistrer le parcours', () => {
      state.pendingCreatedRow = buildCreatedRow(item);
      state.index += 1;
      state.autoResume = state.index < state.exportData.items.length;
      saveState();
      log(state.autoResume ? 'Parcours soumis. Le script reprendra sur la page suivante.' : 'Dernier parcours soumis.');
    });
  }

  // Médiathèque : ouvre "Ajouter un média", sélectionne le dossier Bannières puis une image aléatoire, y compris sur la page suivante si disponible.
  async function selectRandomMedia() {
    const openButton = await waitForElement(SELECTORS.mediaOpen);
    await safeClick(openButton, 'Ouvrir la médiathèque');
    // Certains boutons Drupal nécessitent un second déclenchement souris/pointeur si la modale ne s'ouvre pas.
    if (!(await appears(SELECTORS.mediaDirectory, 2500))) {
      log('La médiathèque ne s’est pas ouverte après le clic standard, nouvel essai avec une séquence complète souris/pointeur.');
      triggerClick(openButton);
      await sleep(state.delayMs);
    }
    const directory = await waitForElement(SELECTORS.mediaDirectory, 15000);
    setSelectValue(directory, DEFAULT_DIRECTORY_VALUE, DEFAULT_DIRECTORY_LABEL);
    const applyFilters = await waitForElement(SELECTORS.mediaApplyFilters, 10000);
    await safeClick(applyFilters, 'Appliquer les filtres de la médiathèque');
    await waitForAjax();

    let choices = await getSelectableMediaChoices();
    const nextPage = findVisibleElement(SELECTORS.mediaNextPage);

  async function insertReturnTrainingLinkInEditor(editor) {
    const text = 'Retour à la page Ma formation';
    const url = 'https://intranet.ac-poitiers.fr/mini-home/ma-formation-0';
    const html = `<p><a href="${url}">${text}</a></p>`;
    editor.focus();
    selectEditorContents(editor);

    if (isFirefox() && await insertReturnTrainingLinkWithToolbar(editor, text, url)) {
      return;
    }

    if (insertReturnTrainingLinkWithCkEditorModel(editor, text, url)) {
      await sleep(250);
      if (editorHasReturnTrainingLink(editor, url)) {
        return;
      }
    }

    dispatchPasteHtml(editor, html, text);
    await sleep(250);
    if (editorHasReturnTrainingLink(editor, url)) {
      return;
    }

    // Firefox peut empêcher l'événement paste sans insérer le HTML : on tente alors l'insertion contenteditable native.
    editor.focus();
    selectEditorContents(editor);
    if (insertHtmlWithExecCommand(html)) {
      await sleep(250);
      if (editorHasReturnTrainingLink(editor, url)) {
        return;
      }
    }

    // Secours final : si CKEditor ne traite ni paste ni execCommand, on écrit uniquement dans CE nouvel éditeur vide.
    editor.innerHTML = html;
    dispatchRichtextInput(editor, html);
    editor.dispatchEvent(new Event('change', { bubbles: true }));
  }

  function insertReturnTrainingLinkWithCkEditorModel(editor, text, url) {
    const ckeditor = findCkEditorInstanceForEditable(editor);
    if (!ckeditor?.model) return false;
    try {
      ckeditor.model.change(writer => {
        const root = ckeditor.model.document.getRoot();
        writer.remove(writer.createRangeIn(root));
        const paragraph = writer.createElement('paragraph');
        writer.append(paragraph, root);
        writer.append(writer.createText(text, { linkHref: url }), paragraph);
      });
      ckeditor.updateSourceElement?.();
      return true;
    } catch (error) {
      console.warn('[EAFC automation] Insertion via modèle CKEditor indisponible, utilisation des secours.', error);
      return false;
    }
  }

  function findCkEditorInstanceForEditable(editable) {
    const instances = pageWindow().Drupal?.CKEditor5Instances;
    if (!instances) return null;
    const values = typeof instances.values === 'function' ? [...instances.values()] : Object.values(instances);
    return values.find(instance => {
      try {
        const viewRoot = instance.editing?.view?.document?.getRoot();
        const domRoot = viewRoot && instance.editing.view.domConverter.mapViewToDom(viewRoot);
        return domRoot === editable || domRoot?.contains(editable) || editable.contains(domRoot);
      } catch (error) {
        return false;
      }
    }) || null;
  }

  function pageWindow() {
    return typeof unsafeWindow === 'undefined' ? window : unsafeWindow;
  }

  function selectEditorContents(editor) {
    const selection = window.getSelection();
    if (!selection) return;
    const range = document.createRange();
    range.selectNodeContents(editor);
    selection.removeAllRanges();
    selection.addRange(range);
  }

  function dispatchPasteHtml(editor, html, text) {
    try {
      const page = pageWindow();
      const clipboardData = new page.DataTransfer();
      clipboardData.setData('text/html', html);
      clipboardData.setData('text/plain', text);
      const pasteEvent = new page.ClipboardEvent('paste', {
        bubbles: true,
        cancelable: true,
        clipboardData,
      });
      editor.dispatchEvent(pasteEvent);
    } catch (error) {
      console.warn('[EAFC automation] Collage CKEditor indisponible, utilisation du secours HTML.', error);
    }
  }

  function editorHasReturnTrainingLink(editor, url) {

  function insertHtmlWithExecCommand(html) {
    try {
      return pageWindow().document.execCommand?.('insertHTML', false, html) || false;
    } catch (error) {
      console.warn('[EAFC automation] execCommand insertHTML indisponible, utilisation du secours HTML.', error);
      return false;
    }
  }

    }
  }

  function findNewCkEditable(existingEditors, scope = document) {
    const scopedEditors = [...scope.querySelectorAll(SELECTORS.ckEditable)].filter(isVisible);
    const editors = scopedEditors.length ? scopedEditors : [...document.querySelectorAll(SELECTORS.ckEditable)].filter(isVisible);
    const newlyCreatedEditors = editors.filter(editor => !existingEditors.has(editor));
    // Le lien doit être posé dans le DERNIER paragraphe créé : si plusieurs CKEditor apparaissent,
    // on privilégie donc le dernier éditeur vide ajouté après le clic "Ajouter Paragraphe".
    return lastEmptyCkEditor(newlyCreatedEditors)
      || newlyCreatedEditors[newlyCreatedEditors.length - 1]
      || lastEmptyCkEditor(editors)
      || editors[editors.length - 1];
  }

  // Autocomplétion Drupal : saisit l'ID demandé, attend vraiment les suggestions, puis laisse Drupal valider la valeur.
  async function fillAutocomplete(input, label) {
    input.focus();
    setValue(input, label);
    input.dispatchEvent(new KeyboardEvent('keydown', { key: 'ArrowDown', bubbles: true }));
    await sleep(Math.max(state.delayMs, MIN_AUTOCOMPLETE_DELAY_MS));
    await waitUntil(() => visibleAutocompleteSuggestions().length > 0, 5000).catch(() => null);
    const suggestions = visibleAutocompleteSuggestions();
    const normalizedLabel = normalize(label);
    const suggestion = suggestions.find(element => {
      const text = normalize(element.textContent);
      return text.includes(normalizedLabel) || text.includes(`(${normalizedLabel})`);
    }) || suggestions[0];
    if (suggestion) {
      await safeClick(suggestion, `Valider l’autocomplétion : ${label}`);
      await waitUntil(() => normalize(input.value) !== normalizedLabel || !isVisible(input), 3000).catch(() => null);
    } else {
      input.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', bubbles: true }));
      input.blur();
    }
    await sleep(Math.max(state.delayMs, MIN_AUTOCOMPLETE_SELECTION_DELAY_MS));
    await waitForAjax();
  }

  // Clic sécurisé : scroll + surbrillance + confirmation optionnelle + séquence souris/pointeur.
  async function safeClick(element, message, beforeClick) {
    if (!element) throw new Error(`Élément introuvable pour : ${message}`);
    element.scrollIntoView({ block: 'center', inline: 'center' });
    element.classList.add('eafc-auto-highlight');
    await sleep(120);
    if (state.safeMode && !confirm(`${message} ?`)) {
      element.classList.remove('eafc-auto-highlight');
      throw new Error(`Action annulée : ${message}`);
    }
    if (beforeClick) beforeClick();
    triggerClick(element);
    element.classList.remove('eafc-auto-highlight');
    await sleep(state.delayMs);
  }

  // Déclenchement robuste d'un clic Drupal sans propriété view (compatible Tampermonkey sandbox).
  function triggerClick(element) {
    element.focus?.({ preventScroll: true });
    ['pointerover', 'pointerenter', 'mouseover', 'mouseenter', 'pointerdown', 'mousedown', 'pointerup', 'mouseup'].forEach(type => {
      element.dispatchEvent(new MouseEvent(type, { bubbles: true, cancelable: true }));
    });
    element.click();
  }

  // Affecte une valeur et déclenche input/change pour que Drupal voie la modification.

    if (!element) return;
    element.focus();
    element.value = value || '';
    element.dispatchEvent(new Event('input', { bubbles: true }));
    element.dispatchEvent(new Event('change', { bubbles: true }));
  }

  // Sélectionne une option par value, sinon par libellé partiel normalisé.
  function setSelectValue(select, value, fallbackLabel) {
    if (!select) return;

      || [...select.options].find(candidate => normalize(candidate.textContent).includes(normalize(fallbackLabel || value)));
    if (!option) throw new Error(`Option introuvable : ${fallbackLabel || value}`);
    select.value = option.value;
    select.dispatchEvent(new Event('change', { bubbles: true }));
  }

      if (paused) throw new Error('Pause demandée');
      const result = predicate();
      if (result) return result;
      await sleep(150);
    }
    throw new Error('Délai dépassé en attendant un élément ou une condition.');
  }

  // Convertit les dates exportées au format JJ/MM/AAAA vers le format input[type=date] AAAA-MM-JJ.
  function toHtmlDate(date) {
    if (!date) return '';
    const match = String(date).match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
    if (match) return `${match[3]}-${match[2]}-${match[1]}`;
    return String(date);
  }

  // Fallback si item.html manque : crée un minimum de HTML compatible fiche formation.
  function wrapFormationHtml(item) {

  function formationRichtextHtml(item) {

  }

  function addFormationUnits(html) {
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const root = doc.body;
    [...root.querySelectorAll('p, li')].forEach(element => {
      const unit = unitForLabelText(element.textContent);
      if (unit) appendUnitToElement(element, unit);
    });
    [...root.querySelectorAll('h1, h2, h3, h4')].forEach(heading => {
      const unit = unitForLabelText(heading.textContent);
      const target = nextContentElement(heading);
      if (unit && target) appendUnitToElement(target, unit);
    });
    return root.innerHTML;
  }

  function unitForLabelText(value) {
    const text = normalize(value);
    if (/^effectif\b/.test(text)) return 'places';
    if (/^(duree totale du parcours|duree du presentiel|duree du distanciel|duree|presentiel|distanciel)\b/.test(text)) return 'heures';
    return '';
  }

  function nextContentElement(element) {
    let next = element.nextElementSibling;
    if (next && /^(H1|H2|H3|H4)$/i.test(next.tagName)) return null;
    while (next && !normalize(next.textContent)) next = next.nextElementSibling;
    return next && !/^(H1|H2|H3|H4)$/i.test(next.tagName) ? next : null;
  }

  function appendUnitToElement(element, unit) {
    const target = lastEditableTextNode(element) || element;
    if (target.nodeType === Node.TEXT_NODE) {
      target.nodeValue = withUnit(target.nodeValue, unit);
      return;
    }
    target.textContent = withUnit(target.textContent, unit);
  }

  function lastEditableTextNode(element) {
    const walker = element.ownerDocument.createTreeWalker(element, NodeFilter.SHOW_TEXT, {
      acceptNode: node => normalize(node.nodeValue) ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT,
    });
    let current;
    let last;
    while ((current = walker.nextNode())) {
      last = current;
    }
    return last;
  }

  function withUnit(value, unit) {
    const text = String(value || '').trimEnd();
    if (!text) return text;
    if (unit === 'heures') {
      const normalizedHours = text.replace(/(\d+(?:[,.]\d+)?)\s*h\b/gi, '$1 heures');
      return /\bheures?\b/i.test(normalizedHours) ? normalizedHours : `${normalizedHours} heures`;
    }
    if (unit === 'places') {
      return /\bplaces?\b/i.test(text) ? text : `${text} places`;
    }
    return text;
  }

  // Extrait en texte lisible les informations de la div .fiche-formation issue de l'export OVP pour la colonne Excel.
  function ficheFormationText(item) {
    const html = formationRichtextHtml(item);
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const fiche = doc.querySelector('.fiche-formation') || doc.body;
    return [...fiche.querySelectorAll('h1, h2, h3, h4, p, li')]
      .map(element => element.textContent.replace(/\s+/g, ' ').trim())
      .filter(Boolean)
      .join('\n');
  }

  function escapeHtml(value) {
    return String(value).replace(/[&<>"]/g, character => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[character]));
  }

  function truncate(value, max) {
    return String(value).slice(0, max);
  }

  // Normalisation utilisée pour les recherches tolérantes (accents, espaces, tirets, casse).
  function normalize(value) {
    return String(value || '')
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')
      .replace(/[–—−]/g, '-')
      .replace(/\s+/g, ' ')
      .trim()
      .toLowerCase();
  }

  function unique(values) {
    return [...new Set(values.map(value => String(value).trim()).filter(Boolean))];
  }

  // Déduplique les objets publics/métiers sans les convertir en "[object Object]".
  function uniquePublics(values) {
    const seen = new Set();
    return values.filter(value => {
      if (!value) return false;
      const key = typeof value === 'string' ? value : value.id || value.path || value.label;
      if (!key || seen.has(key)) return false;
      seen.add(key);
      return true;
    });
  }

  // Transforme un public exporté en chemin Fancytree, par exemple "Tous les personnels > ... > Tous les enseignants".
  function publicPath(value) {
    if (typeof value === 'string') return [value];
    if (value?.path) return String(value.path).split('>').map(part => part.trim()).filter(Boolean);
    return [value?.label || value?.id].filter(Boolean);
  }

  function isVisible(element) {
    if (!element) return false;
    const style = window.getComputedStyle(element);
    return style.display !== 'none' && style.visibility !== 'hidden' && element.getClientRects().length > 0;
  }

  function sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  // Journalise dans la console et dans le panneau flottant.
  function log(message) {
    const line = `[${new Date().toLocaleTimeString()}] ${message}`;
    console.log(`[EAFC automation] ${message}`);
    const output = document.querySelector('#eafc-auto-panel [data-role="log"]');
    if (output) {
      output.textContent = `${line}\n${output.textContent}`.slice(0, 6000);
    }
  }

  // API console utile pour dépanner : window.eafcDrupalFormationAutomation.start/pause/reset/state().
  function exposeDebugApi() {
    window.eafcDrupalFormationAutomation = {
      start: startAutomation,
      pause: () => { paused = true; running = false; },
      reset: () => { state = initialState(); saveState(); },
      state: () => ({ ...state }),
      exportExcel: exportCreatedRowsToExcel,
    };
  }
